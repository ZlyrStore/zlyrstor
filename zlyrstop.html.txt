  <script>
    // BASE DE DATOS INICIAL CON MÍNIMAS IMÁGENES
    const defaultProducts = [
      {
        id: '1',
        title: 'Minecraft Java & Bedrock | Full Access',
        category: 'minecraft',
        price: 7.00,
        desc: 'Acceso total para cambiar correo, clave y nick. Incluye garantía permanente.'
      },
      {
        id: '2',
        title: 'Blood Strike | Cuenta Rango Leyenda + Armas',
        category: 'bloodstrike',
        price: 15.00,
        desc: 'Skins exclusivas, nivel alto y pase de batalla completado.'
      },
      {
        id: '3',
        title: 'Free Fire | Veterana Sakura + MP40',
        category: 'freefire',
        price: 25.00,
        desc: 'Región LATAM. Vinculación limpia lista para el traspaso.'
      },
      {
        id: '4',
        title: 'Roblox | Blox Fruits Max Level',
        category: 'roblox',
        price: 12.00,
        desc: 'Nivel máximo con frutas despertadas e inventario equipado.'
      },
      {
        id: '5',
        title: 'Netflix 1 Mes - Perfil Privado',
        category: 'streaming',
        price: 3.50,
        isUpcoming: true,
        desc: 'Servicio en mantenimiento. Disponible muy pronto.'
      }
    ];

    let products = JSON.parse(localStorage.getItem('zlyr_products_v2')) || defaultProducts;
    let currentCategory = 'inicio';

    function renderProducts() {
      const grid = document.getElementById('productsGrid');
      const search = document.getElementById('searchInput').value.toLowerCase();
      grid.innerHTML = '';

      const filtered = products.filter(p => {
        const matchCategory = currentCategory === 'inicio' || p.category === currentCategory;
        const matchSearch = p.title.toLowerCase().includes(search) || p.desc.toLowerCase().includes(search);
        return matchCategory && matchSearch;
      });

      if (filtered.length === 0) {
        grid.innerHTML = `
          <div class="col-span-full py-12 text-center text-gray-500">
            <i class="fa-solid fa-ghost text-4xl mb-3"></i>
            <p>No hay cuentas registradas en esta categoría.</p>
          </div>
        `;
        return;
      }

      filtered.forEach(p => {
        const isUpcoming = p.isUpcoming || p.category === 'streaming';
        grid.innerHTML += `
          <div class="bg-[#0f131d] rounded-2xl p-5 neon-border transition flex flex-col justify-between relative">
            ${isUpcoming ? `<span class="absolute top-3 right-3 bg-purple-500/20 text-purple-300 text-[10px] font-bold px-2 py-0.5 rounded border border-purple-400/30">Próximamente</span>` : ''}
            <div>
              <div class="flex items-center gap-2 mb-3">
                <span class="text-xs font-bold text-cyan-400 uppercase tracking-wider">${p.category}</span>
              </div>
              <h3 class="font-orbitron font-bold text-white text-base mb-2">${p.title}</h3>
              <p class="text-gray-400 text-xs mb-4 line-clamp-3">${p.desc}</p>
            </div>
            
            <div class="pt-3 border-t border-gray-800/60 flex items-center justify-between mt-auto">
              <div>
                <span class="text-[10px] text-gray-500 uppercase block">Precio</span>
                <span class="font-orbitron text-lg font-bold text-white">$${p.price.toFixed(2)}</span>
              </div>
              ${isUpcoming ? `
                <a href="https://discord.gg/7YQgKtGwH" target="_blank" class="bg-gray-800 hover:bg-gray-700 text-purple-300 border border-purple-500/30 px-3 py-1.5 rounded-lg text-xs font-bold transition">
                  Aviso Discord
                </a>
              ` : `
                <button onclick="openBuyModal('${p.title} - $${p.price.toFixed(2)}')" class="bg-cyan-500/10 hover:bg-cyan-500 text-cyan-400 hover:text-black border border-cyan-500/40 px-3.5 py-1.5 rounded-lg text-xs font-bold transition font-orbitron">
                  Comprar
                </button>
              `}
            </div>
          </div>
        `;
      });
    }

    function switchCategory(cat) {
      currentCategory = cat;
      document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('tab-active'));
      const activeBtn = document.getElementById(`tab-${cat}`);
      if(activeBtn) activeBtn.classList.add('tab-active');

      const hero = document.getElementById('heroSection');
      if (cat === 'inicio') {
        hero.classList.remove('hidden');
      } else {
        hero.classList.add('hidden');
      }

      const titles = {
        inicio: 'Catálogo Disponible',
        minecraft: 'Cuentas de Minecraft',
        bloodstrike: 'Cuentas de Blood Strike',
        freefire: 'Cuentas de Free Fire',
        roblox: 'Cuentas de Roblox',
        streaming: 'Servicios de Streaming'
      };
      document.getElementById('section-title').innerText = titles[cat] || 'Catálogo';
      renderProducts();
    }

    function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
    function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

    function openBuyModal(title) {
      document.getElementById('buyModalProductName').innerText = title;
      openModal('buyModal');
    }

    function handleAddProduct(e) {
      e.preventDefault();
      const title = document.getElementById('prodTitle').value;
      const category = document.getElementById('prodCategory').value;
      const price = parseFloat(document.getElementById('prodPrice').value);
      const desc = document.getElementById('prodDesc').value;

      const newProd = { id: Date.now().toString(), title, category, price, desc, isUpcoming: category === 'streaming' };
      products.unshift(newProd);
      localStorage.setItem('zlyr_products_v2', JSON.stringify(products));
      
      closeModal('adminModal');
      e.target.reset();
      renderProducts();
    }

    renderProducts();
  </script>
